create table faculty(fid int primary key,fname varchar(50),cid int,foreign key (cid) references course(cid));

create table batch(bid int primary key,bname varchar(50),fid int,foreign key (fid) references faculty(fid));

create table Student(sid int primary key,sname varchar(50),bid int,foreign key (bid) references batch(bid));
